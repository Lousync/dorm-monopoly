# v0.5.0 批次 3 实施计划：桌面实体化（物件层 + 手牌 + 可点转盘）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把对局桌面上的关键东西从"贴纸"变成**有厚度、能投影的 3D 实体**——自己的道具变成**手中牌**、现金变成**桌上筹码堆**、体力变成**面前的小件**、转盘变成**可点实体的掷轮入口**——并**拆掉底部操作坞与座位卡上的道具牌位**。

**Architecture:** 批次 2 已把整个 2D 世界贴到一张 3D 平面上；本批次把那张平面**明确降级为"印在桌上的桌垫"**，在 `TableView3D` 里新增一个 `props`（`Node3D`）容器，用**与桌垫同一套 UV 坐标系**给物件定位。点击分两路：落在"桌垫内容"上的照旧送进 `SubViewport`（批次 2 的射线映射），落在**实体物件**上的由一个 `on_table_click` 回调先判——命中转盘/手牌就消费掉。玩法逻辑、`s_state`、`@rpc` 一行不动。

**Tech Stack:** Godot 4.6.2 / GDScript / `gl_compatibility`（实时阴影可用，无 SSAO/景深）。

**Spec:** `doc/development/plans/v0.5.0-桌面实体化-设计.md`（§三 第 1 节 交互重排、§五 第 3 节 3D 物件与房间、§六 批次 3）；反转记录见 `doc/game-design/设计决策留痕.md` §五。

- **不动玩法逻辑与数值**；不动 `doc/development/联机协议.md`——`_broadcast_state` 的 `s_state` 字段**不加不减**；**不动任何 `@rpc` 装饰器与方向**。

- 渲染器固定 `gl_compatibility`。3D 物件必须**有厚度**（`BoxMesh`/`CylinderMesh`），能投影。

```gdscript
class_name TableProps
extends Node3D
## 桌面上的实体物件（见 doc/development/plans/v0.5.0-桌面实体化-设计.md §五）。
## 只负责"摆在哪、长什么样"，不碰任何玩法逻辑。

const WHEEL_RADIUS_PX := 260.0    # 命中半径（棋盘画布像素）；对着图调
const PROPS_Y := 0.02             # 物件离桌垫的高度，避免与桌垫 z-fighting

var _t3: TableView3D
var _wheel_px := Vector2.ZERO

func setup(t3: TableView3D) -> void:
	_t3 = t3

## 在画布坐标 center_px 处立起转盘的实体（有厚度、可投影）。
func build_wheel(center_px: Vector2) -> void:
	_wheel_px = center_px
	var body := MeshInstance3D.new()
	var cyl := CylinderMesh.new()
	cyl.top_radius = 0.62
	cyl.bottom_radius = 0.62
	cyl.height = 0.06
	body.mesh = cyl
	var mat := StandardMaterial3D.new()
	mat.albedo_color = Color(0.16, 0.18, 0.34)
	mat.roughness = 0.7
	body.material_override = mat
	# 放在桌垫上、按画布坐标定位
	var w: Vector3 = _t3.canvas_px_to_world(center_px)
	w.y = _t3.table_mesh.global_position.y + PROPS_Y
	body.global_position = w
	add_child(body)

## 命中判定：画布像素是否落在转盘上。
func wheel_hit(canvas_px: Vector2) -> bool:
	return canvas_px.distance_to(_wheel_px) <= WHEEL_RADIUS_PX
```

`table_props.gd` 追加（**分档做成静态纯函数**，这样能无头单测）：

```gdscript
const CHIP_PER := 5000      # 每 ¥5,000 一枚筹码（对着图调）
const CHIP_MAX := 8         # 筹码堆上限（再多也只画这么多枚）

## 金额 → 筹码枚数（纯函数，可单测）。
static func chip_count(money: int) -> int:
	if money <= 0:
		return 0
	return clampi(int(money / CHIP_PER), 0, CHIP_MAX)

var _chips_root: Node3D
var _stamina_root: Node3D

## 在自己的近端位置摆筹码堆（画布坐标约 (700, 1850)，对着图调）。
func set_chips(money: int) -> void:
	if _chips_root != null and is_instance_valid(_chips_root):
		_chips_root.queue_free()
	_chips_root = Node3D.new()
	add_child(_chips_root)
	var n := chip_count(money)
	var base_px := Vector2(700.0, 1850.0)
	for i in n:
		var chip := MeshInstance3D.new()
		var cyl := CylinderMesh.new()
		cyl.top_radius = 0.10
		cyl.bottom_radius = 0.10
		cyl.height = 0.025
		chip.mesh = cyl
		var mat := StandardMaterial3D.new()
		mat.albedo_color = Color(0.85, 0.66, 0.28)
		chip.material_override = mat
		var w: Vector3 = _t3.canvas_px_to_world(base_px + Vector2(0.0, -float(i) * 14.0))
		w.y = _t3.table_mesh.global_position.y + PROPS_Y + 0.013 + float(i) * 0.026
		chip.global_position = w
		_chips_root.add_child(chip)

## 自己面前一排体力小件：满的亮、用掉的熄灭。
func set_stamina(cur: int, cap: int) -> void:
	if _stamina_root != null and is_instance_valid(_stamina_root):
		_stamina_root.queue_free()
	_stamina_root = Node3D.new()
	add_child(_stamina_root)
	var base_px := Vector2(1150.0, 1850.0)
	for i in cap:
		var pip := MeshInstance3D.new()
		var box := BoxMesh.new()
		box.size = Vector3(0.12, 0.06, 0.12)
		pip.mesh = box
		var mat := StandardMaterial3D.new()
		mat.albedo_color = Color(0.95, 0.78, 0.35) if i < cur else Color(0.22, 0.20, 0.18)
		pip.material_override = mat
		var w: Vector3 = _t3.canvas_px_to_world(base_px + Vector2(float(i) * 40.0, 0.0))
		w.y = _t3.table_mesh.global_position.y + PROPS_Y
		pip.global_position = w
		_stamina_root.add_child(pip)
```

```gdscript
const HAND_MAX := 5
const HAND_CARD_PX := Vector2(220.0, 300.0)   # 单张牌在画布上的占位（对着图调）
const HAND_BASE_PX := Vector2(1024.0, 1780.0) # 手牌中心（近端、偏自己）

var _hand_root: Node3D
var _hand_n := 0

## 按背包重建手中牌；items 每项形如 {"id": "招财猫", "charges": 3}。
func set_hand(items: Array) -> void:
	if _hand_root != null and is_instance_valid(_hand_root):
		_hand_root.queue_free()
	_hand_root = Node3D.new()
	add_child(_hand_root)
	_hand_n = mini(items.size(), HAND_MAX)
	for i in _hand_n:
		var id := String(items[i].get("id", ""))
		var card := MeshInstance3D.new()
		var box := BoxMesh.new()
		box.size = Vector3(0.62, 0.02, 0.86)     # 有厚度的卡
		card.mesh = box
		var mat := StandardMaterial3D.new()
		mat.albedo_color = ItemData.QUALITY_COLORS.get(String(ItemData.def(id).get("quality", "白")),
			Color.WHITE)
		card.material_override = mat
		# 扇形展开：以 HAND_BASE_PX 为中心左右排开
		var off := (float(i) - float(_hand_n - 1) * 0.5) * HAND_CARD_PX.x
		var w: Vector3 = _t3.canvas_px_to_world(HAND_BASE_PX + Vector2(off, 0.0))
		w.y = _t3.table_mesh.global_position.y + PROPS_Y + 0.01
		card.global_position = w
		card.rotation_degrees = Vector3(-14.0, 0.0, -off * 0.012)  # 朝自己倾斜（对着图调）
		_hand_root.add_child(card)

func hand_count() -> int:
	return _hand_n

## 第 i 张牌在画布像素下的矩形（命中判定用）。
func hand_rect(i: int) -> Rect2:
	var off := (float(i) - float(hand_count() - 1) * 0.5) * HAND_CARD_PX.x
	var c := HAND_BASE_PX + Vector2(off, 0.0)
	return Rect2(c - HAND_CARD_PX * 0.5, HAND_CARD_PX)

## 画布像素命中第几张牌；-1 = 没命中。
func hand_hit(canvas_px: Vector2) -> int:
	for i in hand_count():
		if hand_rect(i).has_point(canvas_px):
			return i
	return -1
```

```gdscript
var _hand_slot := -1     # 当前选中的手牌下标（-1 = 未选）

func _on_table_click(canvas_px: Vector2) -> bool:
	if table3d == null or table3d.table_props == null:
		return false
	var tp = table3d.table_props
	# 1) 手牌优先：点牌 = 选中/取消；已选中时再点目标由既有的座位卡/格子逻辑接
	var hi: int = tp.hand_hit(canvas_px)
	if hi >= 0:
		if _hand_slot == hi:
			_hand_slot = -1
			board.set_item_selected(-1)
		else:
			_hand_slot = hi
			board.set_item_selected(hi)
			# 走既有的"选中一张牌"分支：需要目标的两段式道具在此转入选目标态
			_on_item_slot_clicked(my_peer, hi)
		return true
	# 2) 转盘
	if tp.wheel_hit(canvas_px):
		_on_roll_pressed()
		return true
	return false
```
